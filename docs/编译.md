
## 编译

```powershell
cd d:\code\MPA-OpenCl
cmake -S "D:\code\MPA-OpenCl" -B "D:\code\MPA-OpenCl\build" -DCMAKE_BUILD_TYPE=Debug -DOPENSSL_ROOT_DIR="D:/code/vcpkg/installed/x64-windows"
cmake  --build build --config Debug 
```

## 加法

```powershell
cd d:\code\MPA-OpenCl;
cmake  --build build --config Debug --target opencl_ecm_addsub
.\build\Debug\opencl_ecm_addsub.exe -d 1 --bits 512 10000 128 3
```

## 乘法
```powershell
cd d:\code\MPA-OpenCl;
cmake --build build_rel --config Release opencl_ecm_montsqr
.\build_rel\Release\opencl_ecm_montsqr.exe -d 1 --bits 512 5000 128 2   # AMD 设备 1
.\build_rel\Release\opencl_ecm_montsqr.exe -d 0 --bits 512 5000 128 2   # NVIDIA 
.\build_rel\Release\opencl_ecm_montsqr.exe -d 1 --bits 4096 100 128 2
```

## ECM

```powershell
cd d:\code\MPA-OpenCl;
# Release build with vcpkg toolchain
cmake -S "D:\code\MPA-OpenCl" -B "D:\code\MPA-OpenCl\build_rel" -DCMAKE_BUILD_TYPE=Release -DCMAKE_TOOLCHAIN_FILE="D:/code/vcpkg/scripts/buildsystems/vcpkg.cmake" -DOPENSSL_ROOT_DIR="D:/code/vcpkg/installed/x64-windows"
cmake --build build --config Debug --target ecm; cmake --build build_rel --config Release

echo '(2^421-1)' | .\build_rel\Release\ecm.exe -v -d 1 -gpu -sigma 3:1707370477 -gpucurves 64 1e4 0
echo '(2^421-1)' | .\build\Debug\ecm.exe -v -d 1 -sigma 3:1707370477 --go -gpu -gpucurves 64 2000 0
echo '(2^4003-1)' | .\build\Debug\ecm.exe -sigma 3:1876687646 -v -d 1 -gpu -gpucurves 32 1000 0

powershell -NoProfile -ExecutionPolicy Bypass -File tools\test_validate_factors.ps1 2>&1
```
